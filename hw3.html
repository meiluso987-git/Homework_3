Method 1:

#' Q1 Method 1: customers per store(R)
#'
#' Pulls the `customer` table from the sakila SQLite database, then uses
#' base-R `aggregate()` to count customers and sum active customers per`store_id`, and derives the active percentage on the resulting `data.frame`. 
#'
#' @returns A `data.frame` with one row per store and columns
#'   `store_id`, `n` (total customers), `n_active` (customers whose
#'   `active` flag equals '1'), and `pct_active` (100 * n_active / n).
#' @export
#'
#' @examples
#' q1_R()
q1_R <- function() {
  cust <- dbReadTable(con, "customer")          
  
  cust$active_num <- as.integer(cust$active)
  agg <- aggregate(cbind(n = rep(1, nrow(cust)), active_num) ~ store_id,
                   data = cust, FUN = sum)
  names(agg)[names(agg) == "active_num"] <- "n_active"
  agg$pct_active <- 100 * agg$n_active / agg$n
  agg[order(agg$store_id), c("store_id", "n", "n_active", "pct_active")]
}

Method 2:

#' Q1 Method 2: customers per store(SQL)
#'
#' Runs a single `GROUP BY store_id` aggregation against the `customer`table. `active` is stored as `CHAR(1)` ('0'/'1'), so it is cast to `INTEGER` inside SQL before summing.
#'
#' @returns A `data.frame` with one row per store and columns 
#' `store_id`, `n`, `n_active`, and `pct_active`. Identical content to [q1_R()].
#' @export
#'
#' @examples
#' q1_SQL()
q1_SQL <- function() {
dbGetQuery(con, "
    SELECT store_id,
           COUNT(*) AS n,
           SUM(CAST(active AS INTEGER)) AS n_active,
           100.0 * SUM(CAST(active AS INTEGER)) / COUNT(*) AS pct_active
    FROM customer
    GROUP BY store_id
    ORDER BY store_id
  ")
}

b. Staff names and countries

Method 1:

#' Q2 Method 1: staff names and country(R)
#'
#' Pulls four related tables (`staff`, `address`, `city`, `country`) and chains three base-R `merge()` calls to walk the foreign-key path `staff.address_id -> address.city_id -> city.country_id -> country`. No `_list` tables are used.
#'
#' @returns A `data.frame` with one row per staff member and columns
#'   `first_name`, `last_name`, and `country`, ordered by `staff_id`.
#' @export
#'
#' @examples
#' q2_R()
q2_R <- function() {
  staff   <- dbReadTable(con, "staff")
  address <- dbReadTable(con, "address")
  city    <- dbReadTable(con, "city")
  country <- dbReadTable(con, "country")

  out <- merge(staff[, c("staff_id", "first_name", "last_name", "address_id")],
               address[, c("address_id", "city_id")],
               by = "address_id")
  out <- merge(out, city[, c("city_id", "country_id")], by = "city_id")
  out <- merge(out, country[, c("country_id", "country")], by = "country_id")
  out[order(out$staff_id), c("first_name", "last_name", "country")]
}

Method 2:

#' Q2 Method 2: staff names and country(SQL)
#'
#' A single query that `JOIN`s `staff -> address -> city -> country` on the same foreign-key chain used in [q2_R()], letting SQLite do all four-table joins and projection.
#'
#' @returns A `data.frame` with one row per staff member and columns
#'   `first_name`, `last_name`, and `country`, ordered by `staff_id`.
#'   Identical content to [q2_R()].
#' @export
#'
#' @examples
#' q2_SQL()
q2_SQL <- function() {
dbGetQuery(con,"
  SELECT s.first_name, s.last_name, co.country
  FROM staff AS s
  JOIN address AS a ON s.address_id = a.address_id
  JOIN city AS ci ON a.city_id = ci.city_id
  JOIN country AS co ON ci.country_id = co.country_id
  ORDER BY s.staff_id
  ")
}

c. Highest_value films

Method 1:

#' Q3 Method 1: film(s) with highest total rental revenue(R)
#'
#' Pulls `payment`, `rental`, `inventory`, and `film`, then chains
#' `merge()` calls along `payment.rental_id -> rental.inventory_id ->
#' inventory.film_id -> film`. After the joins, `aggregate(amount ~
#' title, FUN = sum)` computes each film's total revenue, and rows
#' whose revenue equals the maximum are kept (so ties are all
#' returned).
#'
#' @returns A `data.frame` with columns `title` and `total_revenue`;
#'   one row for every film whose total rental revenue equals the
#'   across-the-board maximum.
#' @export
#'
#' @examples
#' q3_R()
q3_R <- function() {
  payment   <- dbReadTable(con, "payment")
  rental    <- dbReadTable(con, "rental")
  inventory <- dbReadTable(con, "inventory")
  film      <- dbReadTable(con, "film")

  # payment -> rental on rental_id
  pr <- merge(payment[, c("rental_id", "amount")],
              rental[, c("rental_id", "inventory_id")],
              by = "rental_id")
  pri <- merge(pr, inventory[, c("inventory_id", "film_id")],by = "inventory_id")
  prif <- merge(pri, film[, c("film_id", "title")], by = "film_id")

  rev <- aggregate(amount ~ title, data = prif, FUN = sum)
  names(rev)[2] <- "total_revenue"
  rev[rev$total_revenue == max(rev$total_revenue), ]
}

Method 2:

#' Q3 Method 2: film(s) with highest total rental revenue(SQL)
#'
#' A single query that joins `payment -> rental -> inventory -> film`,
#' groups by film, and keeps only the groups whose `SUM(amount)` equals
#' the maximum group total (computed by a subquery). Using
#' `HAVING ... = (SELECT MAX(s) ...)` rather than `ORDER BY ... LIMIT 1`
#' means every film tied for first place is returned.
#'
#' @returns A `data.frame` with columns `title` and `total_revenue`;
#'   one row for every film whose total rental revenue equals the
#'   across-the-board maximum. Identical content to [q3_R()].
#' @export
#'
#' @examples
#' q3_SQL()
q3_SQL <- function() {
dbGetQuery(con, "
SELECT f.title, SUM(p.amount) AS total_revenue
FROM payment AS p
  JOIN rental AS r ON p.rental_id = r.rental_id
  JOIN inventory AS i ON r.inventory_id = i.inventory_id
  JOIN film AS f ON i.film_id = f.film_id
  GROUP BY f.film_id, f.title
  HAVING SUM(p.amount) = (
    SELECT MAX(s) FROM (
      SELECT SUM(p2.amount) AS s
      FROM payment p2
      JOIN rental AS r2 ON p2.rental_id = r2.rental_id
      JOIN inventory AS i2 ON r2.inventory_id = i2.inventory_id
      GROUP BY i2.film_id
      )
  )
  ORDER BY f.title
")
}

The data-frame approach uses SQL to extract the required tables and then performs the remaining operations in R. The single-SQL approach performs the joins, aggregations, and calculations directly in the database.

The single-SQL approach wins on every question — most dramatically on Q2, where the SQL lets SQLite do three indexed joins in one pass instead of pulling four tables and running three merge() calls in R. Q3’s gap is also large because SQLite streams the payment → rental → inventory → film join + aggregation without ever materialising the intermediate wide table in R.
